        self.sc_weight = sc_weight
        self.knot_dim = knot_dim

    def step(self, batch_x: torch.Tensor, batch_y_noisy: torch.Tensor):
        self.optimizer.zero_grad()

        # Step 1: Forward Pass through Dual-Stream Projection
        # Outputs: z_core (Invariant signal), z_knot (Isolated noise/variance)
